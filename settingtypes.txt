
# Chance of spawning grave. Bigger number means more graves.
# Be really carefull with changing this value, it is easy to get your world fill with graves.
cemeteries_graves_fill_ration (Chance of spawning grave) float 0.0001 0 1


# Chance of spawning cemetery. Bigger number means more cemeteries.

cemeteries_cemeteries_fill_ration (Chance of spawning cemetery) float 0.000005 0 1
